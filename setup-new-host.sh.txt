#!/bin/bash
# RHIDE revival: new-host bootstrap (saved as .txt for easy copying).
# Clones all four repos and builds tvision -> setedit-upstream -> rhide.
# Tested on CachyOS/Arch with gcc 16. Needs: gcc g++ make perl git,
#   ncurses dev, gpm dev, X11 dev libs (Xmu!). On Arch:
#   sudo pacman -S --needed base-devel ncurses gpm libx11 libxmu perl git
set -e
P="$HOME/Projects"
mkdir -p "$P" ~/.local/bin
cd "$P"

echo "=== 1/4 clone (our forks have the revival branches) ==="
[ -d rhide   ] || git clone https://github.com/justaperlhacker/rhide.git
[ -d tvision ] || git clone https://github.com/justaperlhacker/tvision.git
[ -d setedit-upstream ] || git clone https://github.com/set-soft/setedit.git setedit-upstream
[ -d gdb-5.3 ] || git clone https://github.com/justaperlhacker/gdb-5.3.git
# Repair case: dir was cloned from upstream (no modern-gcc there).
# Point origin at our fork, keep upstream for reference.
for r in "tvision justaperlhacker/tvision" "gdb-5.3 justaperlhacker/gdb-5.3"; do
  set -- $r
  url=$(git -C "$1" remote get-url origin 2>/dev/null || true)
  case "$url" in
    *set-soft*)
      git -C "$1" remote rename origin upstream
      git -C "$1" remote add origin "https://github.com/$2.git" ;;
  esac
done
git -C tvision fetch origin
git -C tvision checkout modern-gcc
git -C gdb-5.3 fetch origin
git -C gdb-5.3 checkout modern-gcc

echo "=== 2/4 tvision (static only) ==="
cd "$P/tvision/tvision"
./configure --without-dynamic
make -j"$(nproc)"
ln -sf "$PWD/rhtv-config" ~/.local/bin/rhtv-config

echo "=== 3/4 setedit-upstream libset ==="
cd "$P/setedit-upstream/setedit"
./configure --libset --no-infview \
  --tv-include="$P/tvision/tvision/include" \
  --tv-lib="$P/tvision/tvision/makes"
make needed
make libset

echo "=== 4/4 rhide ==="
cd "$P/rhide"
export TV_INC="$P/tvision/tvision/include" TVOBJ="$P/tvision/tvision/makes" \
       SETSRC="$P/setedit-upstream/setedit" SETOBJ="$P/setedit-upstream/setedit/makes" \
       GDB_SRC="$P/gdb-5.3"
./configure
make -j"$(nproc)"
make rhide
echo
echo "OK: $P/rhide/rhide and $P/rhide/rhgdb/rhgdb.exe built."
echo "Run in terminal with:  env -u DISPLAY $P/rhide/rhide"
echo "Debug builds want -gdwarf-2 (IDE auto-appends it to C debug flags)."
